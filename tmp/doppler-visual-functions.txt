 function draw(){
  const mobile=compact.matches,w=mobile?420:840,h=mobile?400:390;
  const c=cases[index],active=phase!=='reference',evidence=phase==='evidence';
  const size=mobile?.58:.95, road=mobile?280:292;
  const velocity=mobile?15:28, startX=index===1?(mobile?184:370):(mobile?103:180);
  const x=startX+(active?velocity*elapsed:0),listener=index===1?(mobile?45:112):(mobile?365:728);
  const roof=road-135*size,source=x+12*size;
  const youX=index===2?x+57*size:listener,youY=index===2?road-78*size:road-92*(mobile?.78:1);
  let out=`<defs>
  <linearGradient id="sky" x2="0" y2="1"><stop stop-color="#edf5fa"/><stop offset="1" stop-color="#fff"/></linearGradient>
  <linearGradient id="bodypaint" x2="0" y2="1"><stop stop-color="#fff"/><stop offset=".7" stop-color="#edf2f6"/><stop offset="1" stop-color="#d9e3eb"/></linearGradient>
  <linearGradient id="glass" x2="1" y2="1"><stop stop-color="#a5d1e3"/><stop offset="1" stop-color="#4b7b99"/></linearGradient>
  <clipPath id="landscape"><rect width="${w}" height="${road+27}" rx="8"/></clipPath>
  <marker id="motion" markerWidth="7" markerHeight="7" refX="6" refY="3.5" orient="auto"><path d="M0 0L7 3.5L0 7Z" fill="${ink}"/></marker>
  </defs><g clip-path="url(#landscape)">
  <rect width="${w}" height="${road+27}" fill="url(#sky)"/>
  <path d="M0 ${road-37}Q${w*.24} ${road-60} ${w*.46} ${road-34}T${w} ${road-43}V${road+5}H0Z" fill="#deebe8"/>
  <path d="M0 ${road-19}Q${w*.5} ${road-27} ${w} ${road-15}V${road+6}H0Z" fill="#ccdfd8"/>
  <rect y="${road+1}" width="${w}" height="26" fill="#536977"/>
  <path d="M0 ${road+2}H${w}" stroke="#a8bdc5" stroke-width="4"/>
  <path d="M0 ${road+20}H${w}" stroke="#e6e9dc" stroke-width="2" stroke-dasharray="30 24"/>
  `;
  // Sound fronts expand from fixed emission positions, not the moving vehicle.
  const soundSpeed=velocity*10,period=mobile?.52:.48;
  for(let n=0;n<11;n++){
   const age=(elapsed%period)+n*period+.04;
   const center=source-(active?velocity*age:0),radius=soundSpeed*age;
   out+=`<circle cx="${center}" cy="${roof}" r="${radius}" fill="none" stroke="#408ca8" stroke-opacity="${Math.max(.16,.58-n*.035)}" stroke-width="${mobile?1.8:2}"/>`;
  }
  out+='</g>';
  out+=`<rect x="10" y="10" width="${mobile?210:236}" height="34" rx="5" fill="#fff" fill-opacity=".92"/>`+text(21,33,phase==='reference'?'Ambulance at rest':c.name,'start','strong');
  out+=ambulance(x,road,size,index===2);
  if(index!==2)out+=`<g transform="translate(${listener} ${road}) scale(${mobile?.78:1})">${person(0,0)}</g>`;
  if(active){const ay=roof-28;out+=`<path d="M${x-30} ${ay}H${x+42}" fill="none" stroke="${ink}" stroke-width="2.5" marker-end="url(#motion)"/>`;}
  if(index===2){
   out+=line(youX,youY-9,youX+12,youY-57,teal,1.5)+text(youX+15,youY-64,'You','middle','strong',teal);
  }else{
   out+=text(listener,youY-29,'You','middle','strong',teal);
  }
  if(hearing){const label=hearing==='At rest'?'Listening: at rest':'Listening: your sound';out+=text(w/2,road+54,label,'middle','strong',teal);}
  else if(evidence){out+=text(w/2,road+54,`${c.answer==='same'?'Same pitch':c.answer==='higher'?'Higher pitch':'Lower pitch'} · ${Math.round(frequency())} Hz`,'middle','strong',teal);}
  else out+=text(w/2,road+54,'Compare with the siren at rest.','middle','small',gray);
  out+=text(w/2,road+79,evidence?c.detail:phase==='reference'?'Same siren. Same loudness.':index===2?'You ride inside with the siren.':'You stand still beside the road.','middle','small',gray);
  scene.setAttribute('viewBox',`0 0 ${w} ${h}`);scene.innerHTML=out;scene.dataset.phase=phase;scene.dataset.case=String(index);
  scene.setAttribute('aria-label',phase==='reference'?'Ambulance and listener at rest. Circular sound fronts spread from the roof siren.':`${c.name}. ${index===2?'You are seated inside the cab.':'You stand beside the road as the ambulance moves right.'}${evidence?' You hear '+Math.round(frequency())+' Hz instead of the stationary 600 Hz. '+c.detail:''}`);
 }
